## Example

No example of its own: Optional argument of decode() and encode(); the defaults convert every channel in physical units.

It is exercised by:

````cpp title="src/examples/ex_acq_oddball/main.cpp"
while (average[kTarget].count < kEnoughTargets) {
    auto view = reader->lease(1, 2s, maxView);
    if (!view) {
        if (view.error() == AcqErrorCode::EndOfStream) {
            break;
        }
        continue;
    }
    // Native samples to volts in one pass, whatever the device's format and layout.
    const AcqBlockView& samples = **view;
    const std::size_t n = samples.sampleCount();
    if (const AcqError error = decode(samples, AcqMatrixRef<float>::columnMajor(block.data(), frame, n));
        !error.ok()) {
        std::fprintf(stderr, "Cannot decode %s: %s\n", uri.c_str(), error.text().c_str());
        break; // the lease returns the view
    }
    std::uint64_t s = samples.firstSample();
    for (std::size_t i = 0; i < n; ++i, ++s) {
        std::memcpy(&history[(s & (capacity - 1)) * frame], &block[i * frame], frame * sizeof(float));
    }
    for (std::size_t e = 0; e < samples.events().size(); ++e) {
        const AcqEvent& event = samples.events()[e];
        if ((event.code == kStandard || event.code == kTarget) && event.sample >= pre) {
            pending.push_back(event);
        }
    }
    received = s;
    if (!view->release().ok()) {
        std::fprintf(stderr, "The reader fell behind; epochs over the lost samples are skipped\n");
    }
    // An epoch is complete once its last post-stimulus sample has arrived.
    auto complete = [&](const AcqEvent& event) {
        return event.sample + post <= received;
    };
    for (const AcqEvent& event : pending) {
        if (complete(event)) {
            cutEpoch(event);
        }
    }
    pending.erase(std::remove_if(pending.begin(), pending.end(), complete), pending.end());
}
````
