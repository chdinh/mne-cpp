```cpp title="src/examples/ex_mne_api/main.cpp"
// MNE-C style reader: buffer directory, channel selection, projection and overlap-add filter
MNEFilterDef filter; // 40 Hz lowpass, 5 Hz transition, 4096-sample blocks
filter.filter_on = true;
filter.size = 4096;
filter.taper_size = 2048;
filter.lowpass = filter.eog_lowpass = 40.0f;
filter.lowpass_width = filter.eog_lowpass_width = 5.0f;
std::unique_ptr<MNERawData> mneRaw(MNERawData::open_file(rawFile.fileName().toStdU16String(), false, false, filter));
const MNERawInfo& rawInfo = *mneRaw->info;
const MNERawBufDef& firstBuffer = mneRaw->bufs.front();

MNEChSelection sel; // channel names are stored without spaces
sel.chspick = sel.chspick_nospace = {"MEG 0113", "EEG 001", "EOG 061"};
sel.pick_deriv = VectorXi::Constant(sel.nchan(), -1);
sel.pick.resize(sel.nchan());
for (int c = 0; c < sel.nchan(); ++c) {
    sel.pick[c] = static_cast<int>(BASELIB::indexOf(mneRaw->ch_names, sel.chspick[c]));
}

const int ns = 25;
MNERawData::RowMajorMatrixXf segment(sel.nchan(), ns);
mneRaw->pick_data(&sel, mneRaw->first_samp + 2990, segment); // calibrated, across a buffer boundary
const double eegRaw = segment(1, 0);
mneRaw->pick_data_filt(&sel, mneRaw->first_samp + 2990, segment);
const double eegFilt = segment(1, 0);

// The file stores its 4 SSP projectors as inactive; make_projection switches them on
std::unique_ptr<MNEProjOp> proj = MNEProjOp::make_projection({rawFile.fileName().toStdU16String()}, rawInfo.chInfo).value();
proj->assign_channels(mneRaw->ch_names);
const bool compiled = proj->make_proj();
const MNEProjItem& firstItem = proj->items[0];
mneRaw->proj = std::move(proj);
mneRaw->pick_data_proj(&sel, mneRaw->first_samp + 2990, segment);
const double eegProj = segment(1, 0);
```

### assign_channels(chs, deriv)

Match the defined channels against the channels of a recording (MNE-C `mne_ch_selection_assign_chs` and `mne_ch_selection_assign_chs_info`).

A definition matches a channel name ignoring case, with or without the definition's own spaces, and also the part of a channel name before a dash. Definitions without such a channel are looked up among the valid derived channels.

**Parameters:**

- **chs** : *std::span&lt; const [FiffChInfo](/docs/api/fiff/fiff-ch-info) &gt;*
  Channels of the recording.

- **deriv** : *const [MNEDeriv](/docs/api/mne/mne-deriv) **
  Derivations matched to the recording, or nullptr.

**Returns:**

- *int* — The number of definitions matched to a recorded channel (derivations excluded).

### these(selname, names)

Create a user selection of the given channel names (MNE-C `mne_ch_selection_these`).

**Parameters:**

- **selname** : *std::string*
  Name of the selection.

- **names** : *std::span&lt; const std::string &gt;*
  Channel names.

**Returns:**

- *MNEChSelection* — The selection; call `assign_channels()` before picking with it.
